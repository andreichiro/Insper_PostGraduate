SQL utilizada
SELECT tn.tag, COUNT(*) AS freq
        FROM tags t
        JOIN tag_names tn ON tn.tid = t.tid
        GROUP BY tn.tag
        ORDER BY freq DESC, tn.tag
        LIMIT 25
